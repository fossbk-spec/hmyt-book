# Chương 3 Code Lab: Hồi quy Tuyến tính, Logistic & Odds Ratio (95% CI)
import numpy as np
import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import roc_auc_score

# Mô phỏng tính Odds Ratio
np.random.seed(42)
N = 1000
age = np.random.normal(55, 10, N)
sbp = np.random.normal(130, 15, N)
z = -6.0 + 0.04 * age + 0.03 * sbp
prob = 1 / (1 + np.exp(-z))
y = (np.random.rand(N) < prob).astype(int)

clf = LogisticRegression()
clf.fit(np.column_stack([age, sbp]), y)
odds_ratios = np.exp(clf.coef_[0])
print(f"Odds Ratio Tuổi: {odds_ratios[0]:.4f}, Odds Ratio Huyết áp tâm thu: {odds_ratios[1]:.4f}")

# Chương 5 Code Lab: Phân loại Ung thư vú (WDBC) với RFE và RBF-SVM
from sklearn.datasets import load_breast_cancer
from sklearn.feature_selection import RFE
from sklearn.svm import SVC
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

X, y = load_breast_cancer(return_X_y=True)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, stratify=y, random_state=42)
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

svm = SVC(kernel='linear')
rfe = RFE(estimator=svm, n_features_to_select=10)
X_train_rfe = rfe.fit_transform(X_train_scaled, y_train)
X_test_rfe = rfe.transform(X_test_scaled)

rbf_svm = SVC(kernel='rbf', C=100.0, gamma=0.001)
rbf_svm.fit(X_train_rfe, y_train)
print(f"Độ chính xác trên tập Test: {rbf_svm.score(X_test_rfe, y_test):.4f}")
